<script>
    let healthMode = 'who';
    let currentView = 'map';
    let activeSensor = null;
    let hCharts = {}, sCharts = {};

    const standards = {
        who: { no2: { f: 1.88, g: 40, m: 90, max: 200 }, co: { f: 1.15, g: 5, m: 7.5, max: 15 }, nh3: { f: 700, g: 270, m: 350, max: 1000 } },
        eu: { no2: { f: 1.88, g: 120, m: 200, max: 400 }, co: { f: 1.15, g: 10, m: 15, max: 25 }, nh3: { f: 700, g: 500, m: 1000, max: 14000 } }
    };

    function createChart(ctxId, pollutant, isHealth) {
        const ctx = document.getElementById(ctxId).getContext('2d');
        const c = standards[healthMode][pollutant];
        return new Chart(ctx, {
            type: 'line',
            data: { labels: [], datasets: [{ label: pollutant.toUpperCase(), data: [], borderColor: '#003399', tension: 0.3 }] },
            options: {
                plugins: {
                    annotation: {
                        annotations: {
                            good: { type: 'box', yMin: 0, yMax: c.g, backgroundColor: 'rgba(80, 240, 230, 0.3)', borderWidth: 0 },
                            fair: { type: 'box', yMin: c.g, yMax: c.m, backgroundColor: 'rgba(240, 230, 60, 0.3)', borderWidth: 0 },
                            poor: { type: 'box', yMin: c.m, yMax: c.max, backgroundColor: 'rgba(255, 80, 80, 0.3)', borderWidth: 0 }
                        }
                    },
                    legend: { display: false }
                },
                scales: { y: { beginAtZero: true, max: c.max } }
            }
        });
    }

    // Initialize all Charts
    ['no2', 'co', 'nh3'].forEach(p => {
        hCharts[p] = createChart(`chart-h-${p}`, p, true);
        sCharts[p] = createChart(`chart-s-${p}`, p, false);
    });

    function switchView(view) {
        currentView = view;
        document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
        document.getElementById(`view-${view}`).classList.add('active');
        document.querySelectorAll('.nav-links button').forEach(b => b.classList.remove('active'));
        document.getElementById(`link-${view}`).classList.add('active');
        closeSidePanel();
    }

    function openSidePanel(sid) {
        activeSensor = sid;
        document.getElementById('side-title').innerText = "Sensor: " + sid;
        document.getElementById('side-panel').classList.add('open');
    }

    function closeSidePanel() {
        document.getElementById('side-panel').classList.remove('open');
        activeSensor = null;
    }

    function setHealthMode(m) {
        healthMode = m;
        document.querySelectorAll('.tgl-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(`mode-${m}`).classList.add('active');
        // Update all chart ranges
        [hCharts, sCharts].forEach(group => {
            for (let p in group) {
                const c = standards[m][p];
                group[p].options.scales.y.max = c.max;
                group[p].options.plugins.annotation.annotations.good.yMax = c.g;
                group[p].options.plugins.annotation.annotations.fair.yMin = c.g;
                group[p].options.plugins.annotation.annotations.fair.yMax = c.m;
                group[p].options.plugins.annotation.annotations.poor.yMin = c.m;
                group[p].options.plugins.annotation.annotations.poor.yMax = c.max;
                group[p].update();
            }
        });
    }

    function filterHealth(ids, cardId) {
        document.querySelectorAll('.card').forEach(c => c.classList.remove('active'));
        document.getElementById(cardId).classList.add('active');
        document.querySelectorAll('.graph-card').forEach(g => g.classList.remove('visible'));
        ids.forEach(id => document.getElementById(`h-${id}`).classList.add('visible'));
    }

    async function tick() {
        // Update Health Dashboard (Mean)
        if (currentView === 'health') {
            const r = await fetch('/api/mean');
            const data = await r.json();
            if (data.length) updateCharts(hCharts, data);
        }
        // Update Map Side Panel (Specific Sensor)
        if (activeSensor) {
            const r = await fetch(`/api/sensor/${activeSensor}`);
            const data = await r.json();
            if (data.length) updateCharts(sCharts, data);
        }
    }

    function updateCharts(chartGroup, logs) {
        const times = logs.map(l => l.time);
        for (let p in chartGroup) {
            const f = standards[healthMode][p].f;
            chartGroup[p].data.labels = times;
            chartGroup[p].data.datasets[0].data = logs.map(l => (l[p] * f).toFixed(1));
            chartGroup[p].update('none');
        }
    }

    setInterval(tick, 3000);
</script>
